def detect_encoding(file_path):
    """自动检测文件的编码格式（兼容 crdownload 临时文件）"""
    try:
        with open(file_path, 'rb') as f:
            # 读取更多数据提高检测准确性（分片文件可能不完整）
            raw_data = f.read(50000)
            if len(raw_data) == 0:
                print(f"警告：文件 {file_path} 为空")
                return 'utf-8'
            result = chardet.detect(raw_data)
            encoding = result['encoding']
            # 处理检测结果为空的情况
            if encoding is None:
                encoding = 'utf-16'          # 兜底默认编码
            print(f"检测到文件编码: {encoding}, 置信度: {result['confidence']:.2f}")
            return encoding
    except Exception as e:
        print(f"检测编码失败: {str(e)}，使用默认编码 utf-16")
        return 'utf-16'

def classify_products(df, top20_dict):
    """
        始终在榜：有数据的年份里，每年都在本品类榜内
        曾经在榜：至少一年在榜，但不是所有年份都在榜
        从未在榜：从未进入本品类榜内
    """
    # 按 产品-国家-品类 维度去重
    all_units = df[['Unified Name', '国家', '合并后品类']].drop_duplicates()
    years = ['2021', '2022', '2023', '2024', '2025']

    for _, row in all_units.iterrows():
        product_name, country, genre = row['Unified Name'], row['国家'], row['合并后品类']
        yearly_status, total_years_in_top20 = {}, 0

        for year in years:
            # 本品类内下载或收入任一进入前 20 即算在榜
            in_dl  = product_name in top20_dict['download'].get((year, country, genre), [])
            in_rev = product_name in top20_dict['revenue'].get((year, country, genre), [])
            if in_dl or in_rev:
                yearly_status[year] = '在榜'
                total_years_in_top20 += 1
            else:
                yearly_status[year] = '不在榜'

        # 以"有数据的年份"为分母，避免缺数据被误判为掉榜
        years_with_data = sum(1 for v in yearly_status.values() if v != '无数据')
        if years_with_data > 0 and total_years_in_top20 == years_with_data:
            classification = '始终在榜'
        elif total_years_in_top20 > 0:
            classification = '曾经在榜'
        else:
            classification = '从未在榜'

def create_session():
    """创建带重试机制的 requests 会话"""
    session = requests.Session()
    retry_strategy = Retry(
        total=3,                                     # 总重试次数
        backoff_factor=1,                            # 重试间隔系数
        status_forcelist=[412, 500, 502, 503, 504],  # 需要重试的状态码
        allowed_methods=["GET"]
    )
    adapter = HTTPAdapter(max_retries=retry_strategy)
    session.mount("https://", adapter)
    session.headers.update(HEADERS)                  # 模拟真实浏览器
    return session
