def classify_products(df, top20_dict):
    """
        始终在榜：有数据的年份里，每年都在本品类榜内
        曾经在榜：至少一年在榜，但不是所有年份都在榜
        从未在榜：从未进入本品类榜内
    """
    # 按 产品-国家-品类 维度去重
    all_units = df[['Unified Name', '国家', '合并后品类']].drop_duplicates()
    years = ['2021', '2022', '2023', '2024', '2025']

    for _, row in all_units.iterrows():
        product_name, country, genre = row['Unified Name'], row['国家'], row['合并后品类']
        yearly_status, total_years_in_top20 = {}, 0

        for year in years:
            # 本品类内下载或收入任一进入前 20 即算在榜
            in_dl  = product_name in top20_dict['download'].get((year, country, genre), [])
            in_rev = product_name in top20_dict['revenue'].get((year, country, genre), [])
            if in_dl or in_rev:
                yearly_status[year] = '在榜'
                total_years_in_top20 += 1
            else:
                yearly_status[year] = '不在榜'

        # 以"有数据的年份"为分母，避免缺数据被误判为掉榜
        years_with_data = sum(1 for v in yearly_status.values() if v != '无数据')
        if years_with_data > 0 and total_years_in_top20 == years_with_data:
            classification = '始终在榜'
        elif total_years_in_top20 > 0:
            classification = '曾经在榜'
        else:
            classification = '从未在榜'

def calculate_statistics(df_clean, category_name, category_col):
    """核心统计函数：基于"已填写"字段判断分母"""
    metrics = {'样本用户属性': category_name}

    df_clean['是否愿意']  = df_clean['是否愿意'].apply(safe_string_convert)
    df_clean['1试玩意愿'] = df_clean['1试玩意愿'].apply(safe_string_convert)

    # 分母口径：有效爱好用户 + 已查看未填写用户
    hobby_users    = df_clean[df_clean[category_col] == '有出于爱好体验过']
    willing_no_choice = df_clean[(df_clean['是否愿意'] == '愿意') &
                                (df_clean['1试玩意愿'].isin(['', '-']))]
    sample_users = pd.concat([hobby_users, willing_no_choice]).drop_duplicates(subset=['序号'])
    metrics['样本量'] = len(sample_users)

    # 整体下载意愿
    hobby_willing = hobby_users[hobby_users['1试玩意愿'] == '愿意试玩']
    metrics['整体下载意愿'] = (f'{len(hobby_willing) / metrics["样本量"]:.0%}'
                              if metrics['样本量'] else '0%')

    # 尖叫度：仅在"愿意试玩"子集内取均值
    df_clean['1尖叫度'] = df_clean['1尖叫度'].apply(safe_string_convert)
    valid_scream = hobby_willing[hobby_willing['1尖叫度'] != '-']['1尖叫度']
    scream_values = []
    for val in valid_scream:
        try:
            scream_values.append(float(val))
        except (ValueError, TypeError):
            continue
    metrics['尖叫度-愿意下载'] = f'{np.mean(scream_values):.2f}' if scream_values else '0.00'

    # 游玩时长：分母 = "已填写"字段为正向值的样本
    duration_col = auto_detect_play_duration_column(df_clean.columns)
    filled_col   = auto_detect_filled_column(df_clean.columns)

async def get_up_videos_in_range(uid, start_date, end_date, credential=None):
    """获取指定时间范围内单个 UP 主的投稿"""
    u = user.User(uid=uid, credential=credential)
    up_name = (await u.get_user_info())["name"]
    all_videos = []

    while True:
        videos = await u.get_videos()            # 官方库分页拉取
        if not videos.get("list", {}).get("vlist"):
            break                                # 没有更多视频
        for v in videos["list"]["vlist"]:
            pub_date = datetime.fromtimestamp(v["created"])
            if start <= pub_date <= end:
                all_videos.append({
                    "up主名称": up_name,
                    "视频标题": v["title"],       # 标题
                    "视频链接": f"https://www.bilibili.com/video/{v['bvid']}",
                    "发布时间": pub_date.strftime("%Y-%m-%d %H:%M:%S")
                })
            elif pub_date < start:
                # 发布时间早于起始时间，后续都更早，提前退出
                return all_videos
    return all_videos
